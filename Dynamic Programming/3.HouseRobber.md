# 198. House Robber 1
### Memoization approach
```cpp
class Solution {
public:
    int recurse(vector<int>&nums,int index,vector<int> &dp){
        
        if(index == 0){
            return nums[index];
        }
        if(index < 0) return 0;
        if(dp[index]!=-1) return dp[index];
        int take = nums[index] + recurse(nums,index-2,dp);
        int nottake = recurse(nums,index-1,dp);
        return dp[index] = max(take,nottake);

    }
    int rob(vector<int>& nums) {
        vector<int> dp(nums.size(),-1);
        return recurse(nums,nums.size()-1,dp);
       
    }
};
```

### Tabulation approach
```cpp
class Solution {
public:
    int dp(vector<int>& nums,int index,vector<int> &mem){
        if(index == nums.size() -1) return nums[index];
        if(index >= nums.size())return 0;
        if(mem[index]!=-1) return mem[index];
        int take = nums[index]+dp(nums,index+2,mem);
        int nottake = dp(nums,index+1,mem);
        return mem[index] = max(take,nottake);
    }
    int rob(vector<int>& nums) {
        vector<int> mem(nums.size(),-1);
       return dp(nums,0,mem);

    }
};
```

## 213. House Robber II

### Memoization
```cpp
class Solution {
public:
    int backtrack(vector<int>&nums,int index,bool first,vector<int> &dp){
        if(index == nums.size()-1){
            if(!first) return nums[index];
            return 0;
        }
        if(index >= nums.size()) return 0;
    if(dp[index]!=-1) return dp[index];
        int take = nums[index] + backtrack(nums,index+2,first,dp);
        int nottake = backtrack(nums,index+1,first,dp);
        return dp[index] = max(take,nottake);
    }
    int rob(vector<int>& nums) {
        // int index = nums.size()-1;
        if(nums.size() == 1) return nums[0];
        vector<int> dp1(nums.size(),-1);
        vector<int> dp2(nums.size(),-1);
        int take = backtrack(nums,0,true,dp1);
        int nottake = backtrack(nums,1,false,dp2);
        return max(take,nottake);
    }
};
```

### Tabulation
```cpp
class Solution {
public:
    int tabulation(vector<int> &nums){
        vector<int> dp(nums.size()+1,0);
        dp[0] = 0; //no house till now
        dp[1] = nums[0]; //only one house till now
        for(int i=2 ;i<= nums.size() ;i++){
            int take = nums[i-1];
            if(i>2) take += dp[i-2];
            int nottake = dp[i-1];
            dp[i] = max(take,nottake);
        }
        return dp[nums.size()];
    }
    int rob(vector<int>& nums) {
        if(nums.size()==1) return nums[0];
        vector<int> temp1;
        vector<int> temp2;
        for(int i = 0 ; i < nums.size()-1 ; i++){
            temp1.push_back(nums[i]);
        }
        for(int i=1 ; i <nums.size();i++){
            temp2.push_back(nums[i]);
        }
        return max(tabulation(temp1),tabulation(temp2));
    }
};
```
