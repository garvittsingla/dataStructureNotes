# 260. Single Number III


```cpp
class Solution {
public:
    vector<int> singleNumber(vector<int>& nums) {
        if(nums.size() == 2) return nums;
        int xorans = 0;
        for(int i = 0 ; i < nums.size() ;i++){
            xorans ^= nums[i];
        }

        //find the first different bit
        int bit = 0;
        while(bit != 31){
            //check the ith bit of xor
            if((xorans & (1<<bit))!= 0 ) break;
            else bit++;
        }
        //dividing in the groups according to that bit
        vector<int> setbit; //that bit is set
        vector<int> notset;
        for(int i = 0 ; i < nums.size() ;i++){
            if((nums[i] & (1 << bit))!= 0) setbit.push_back(nums[i]);
            else notset.push_back(nums[i]);
        }
        int ans1 = 0;
        for(int i = 0 ; i < setbit.size();i++){
            ans1 ^= setbit[i];
        }
        int ans2 = 0;
        for(int i = 0 ; i < notset.size();i++){
            ans2 ^= notset[i];
        }

        return {ans1,ans2};
    }
};
```

We will find the xor of all the numbers and then find the bit that is different in both the numbers , and divide all the number in those 2 groups wether the bit is set or not, after finding the xor again, will find the numbers
