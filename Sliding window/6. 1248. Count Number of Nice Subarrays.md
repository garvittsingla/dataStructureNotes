# LC. 1248. Count Number of Nice Subarrays

## Approach : The same question of counting the binary subarray with sum, just convert the humber to 0 or 1 to get the odd or even track
```cpp
class Solution {
public:
    int numberOfSubarrays(vector<int>& nums, int goal) {
        for(int i = 0 ; i < nums.size();i++){
            if(nums[i]%2 == 0) nums[i]=0;
            else nums[i] = 1;
        }
        int ans= 0;
        unordered_map<int,int> mpp; //to track the sum which happened the last time
        int sum = 0;
        for(int i = 0 ; i < nums.size();i++){
            sum += nums[i];
            if(goal == sum) ans++;
            int rem = sum - goal;
            if(mpp.count(rem)){
                ans += mpp[rem];
            }
            mpp[sum]++; //update the freq everytime , so that we can check 
            
        }
        return ans;
    }
};
```
