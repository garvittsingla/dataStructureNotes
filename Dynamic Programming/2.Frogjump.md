# 403. Frog Jump

class Solution {
public:     
    bool dp(vector<int>& nums,map<pair<int,int>,bool>& dpset,int index,int k){
        if(index == nums.size()-1) return true;
        pair<int,int> p = {index,k};
        if(dpset.count(p)){
            return dpset[p];
        }
        for(int i = index+1 ; i < nums.size() ;i++){
            if(nums[i]-nums[index]==k){
                bool val = dp(nums,dpset,i,k);
                if(val){a
                    dpset[p] = true;
                    return true;
                } 
            }else if(nums[i]-nums[index]==k+1){
                bool val = dp(nums,dpset,i,k+1);
                if(val){
                    dpset[p] = true;
                    return true;
                } 
            }else if(nums[i]-nums[index]==k-1){
                bool val = dp(nums,dpset,i,k-1);
                if(val){
                    dpset[p] = true;
                    return true;
                } 
            }
        }
        dpset[p] = false;
        return false;
    }
    bool canCross(vector<int>& stones) {
        map<pair<int,int>,bool> dpset;
        if(stones[1]>1) return false;
        return dp(stones,dpset,1,1);
    }
};
